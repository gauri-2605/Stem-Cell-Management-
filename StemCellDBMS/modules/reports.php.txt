// Fetch summary data
$total_patients = mysqli_fetch_assoc(mysqli_query($conn, "SELECT COUNT(*) AS total FROM patients"))['total'];

$total_banks = mysqli_fetch_assoc(mysqli_query($conn, "SELECT COUNT(*) AS total FROM stemcell_banks"))['total'];

    $pdf->Ln(10);

    $pdf->Cell(0,10,"Generated on: ".date("Y-m-d H:i:s"),0,1);
    $pdf->Output("D", "StemCell_Report.pdf");
    exit;
}
